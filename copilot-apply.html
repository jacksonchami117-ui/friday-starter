import os, shutil, json, subprocess, uuid
from pathlib import Path
from typing import List, Dict, Callable

FFMPEG = "ffmpeg"
FONT = "/usr/share/fonts/truetype/dejavu/DejaVuSans.ttf"

def _run(cmd: List[str], on_progress: Callable[[int], None] = None):
    proc = subprocess.Popen(cmd, stdout=subprocess.PIPE, stderr=subprocess.STDOUT, text=True)
    for _line in proc.stdout:
        pass
    return proc.wait()

def _ensure_dir(p: Path):
    p.mkdir(parents=True, exist_ok=True)

def render_manifest(manifest: Dict, workdir: Path, output_path: Path, thumb_path: Path, progress_cb: Callable[[int], None]):
    _ensure_dir(workdir)
    tmp = workdir / f"tmp-{uuid.uuid4().hex}"
    _ensure_dir(tmp)
    try:
        segments = [s for s in manifest.get("segments", []) if s.get("type") == "video"]
        overlays = [s for s in manifest.get("segments", []) if s.get("type") == "text"]

        part_files = []
        for idx, seg in enumerate(segments):
            src = seg.get("src")
            if not src:
                raise RuntimeError(f"Segment {idx} missing 'src'")
            start = str(seg.get("start", 0))
            end = seg.get("end")
            in_args = ["-ss", start] if float(start) > 0 else []
            if end is not None:
                in_args += ["-to", str(float(end) - float(start))]
            in_file = str(src)
            part = tmp / f"part_{idx:03d}.mp4"
            part_cmd = [FFMPEG, "-y"] + in_args + ["-i", in_file,
                         "-an",
                         "-c:v", "libx264", "-preset", "veryfast", "-crf", "23",
                         "-pix_fmt", "yuv420p",
                         str(part)]
            rc = _run(part_cmd)
            if rc != 0:
                raise RuntimeError(f"FFmpeg failed on segment {idx}")
            part_files.append(part)
            progress_cb(min(20, 2 + int((idx + 1) / max(1, len(segments)) * 15)))

        concat_list = tmp / "concat.txt"
        with concat_list.open("w") as f:
            for p in part_files:
                f.write(f"file '{p.as_posix()}'\n")

        base_mp4 = tmp / "base.mp4"
        concat_cmd = [FFMPEG, "-y", "-f", "concat", "-safe", "0",
                      "-i", str(concat_list),
                      "-c:v", "libx264", "-preset", "veryfast", "-crf", "23",
                      "-pix_fmt", "yuv420p",
                      str(base_mp4)]
        rc = _run(concat_cmd)
        if rc != 0:
            raise RuntimeError("FFmpeg concat failed")
        progress_cb(40)

        final_src = base_mp4
        if overlays:
            draw_filters = []
            for o in overlays:
                txt = (o.get("text") or "").replace(":", "\\:").replace("'", "\\'")
                x = o.get("x", "(w-text_w)/2")
                y = o.get("y", "(h-text_h)-50")
                start = float(o.get("start", 0))
                end = o.get("end")
                fontsize = int(o.get("fontsize", 48))
                fontcolor = o.get("fontcolor", "white")
                enable = f"between(t,{start},{end})" if end is not None else f"gte(t,{start})"
                draw_filters.append(
                    f"drawtext=fontfile='{FONT}':text='{txt}':x={x}:y={y}:fontsize={fontsize}:fontcolor={fontcolor}:enable='{enable}'"
                )
            vf = ",".join(draw_filters)
            out_with_text = tmp / "with_text.mp4"
            text_cmd = [FFMPEG, "-y", "-i", str(base_mp4), "-vf", vf,
                        "-c:v", "libx264", "-preset", "veryfast", "-crf", "23",
                        "-pix_fmt", "yuv420p", "-c:a", "copy",
                        str(out_with_text)]
            rc = _run(text_cmd)
            if rc != 0:
                raise RuntimeError("FFmpeg overlay failed")
            final_src = out_with_text
        progress_cb(70)

        output_path.parent.mkdir(parents=True, exist_ok=True)
        shutil.move(str(final_src), str(output_path))
        progress_cb(85)

        thumb_cmd = [FFMPEG, "-y", "-ss", "00:00:01", "-i", str(output_path), "-frames:v", "1", str(thumb_path)]
        rc = _run(thumb_cmd)
        if rc != 0:
            thumb_cmd = [FFMPEG, "-y", "-i", str(output_path), "-frames:v", "1", str(thumb_path)]
            _run(thumb_cmd)
        progress_cb(100)
    finally:
        shutil.rmtree(tmp, ignore_errors=True)

import os, json, requests

SENDGRID_API_KEY = os.getenv("SENDGRID_API_KEY")
SENDGRID_FROM = os.getenv("SENDGRID_FROM")
SENDGRID_TO = os.getenv("SENDGRID_TO")

TWILIO_SID = os.getenv("TWILIO_SID")
TWILIO_AUTH = os.getenv("TWILIO_AUTH")
TWILIO_FROM = os.getenv("TWILIO_FROM")
TWILIO_TO = os.getenv("TWILIO_TO")

def email(subject: str, body: str):
    if not (SENDGRID_API_KEY and SENDGRID_FROM and SENDGRID_TO):
        return False
    url = "https://api.sendgrid.com/v3/mail/send"
    headers = {"Authorization": f"Bearer {SENDGRID_API_KEY}", "Content-Type": "application/json"}
    data = {
        "personalizations": [{"to": [{"email": SENDGRID_TO}], "subject": subject}],
        "from": {"email": SENDGRID_FROM},
        "content": [{"type": "text/plain", "value": body}],
    }
    r = requests.post(url, headers=headers, data=json.dumps(data), timeout=10)
    return r.status_code in (200, 202)

def sms(body: str):
    if not (TWILIO_SID and TWILIO_AUTH and TWILIO_FROM and TWILIO_TO):
        return False
    url = f"https://api.twilio.com/2010-04-01/Accounts/{TWILIO_SID}/Messages.json"
    data = {"From": TWILIO_FROM, "To": TWILIO_TO, "Body": body}
    r = requests.post(url, data=data, auth=(TWILIO_SID, TWILIO_AUTH), timeout=10)
    return r.status_code in (200, 201)

#!/usr/bin/env python3
import os, time, json, requests

BASE = os.environ.get("BASE_URL", "http://localhost:10000")

def login(sess):
  r = sess.get(f"{BASE}/auth/login")
  assert r.status_code == 200
  r = sess.post(f"{BASE}/auth/login", data={"password": os.environ.get("ADMIN_PASSWORD","admin")}, allow_redirects=False)
  assert r.status_code in (302, 303)

def metrics(sess):
  r = sess.get(f"{BASE}/metrics"); assert r.ok

def editor_upload(sess):
  # generate a tiny color clip using ffmpeg, then upload
  import subprocess, tempfile
  tmpdir = tempfile.mkdtemp()
  p1 = os.path.join(tmpdir, "a.mp4")
  cmd = ["ffmpeg","-y","-f","lavfi","-i","color=c=blue:s=320x240:d=1","-c:v","libx264","-t","1",p1]
  subprocess.check_call(cmd)
  with open(p1, "rb") as f:
    r = sess.post(f"{BASE}/editor/upload", files={"file": f})
    assert r.ok
    return r.json()["path"]

def render_job(sess, path):
  payload = {"manifest": {"version":1, "segments":[{"type":"video","src":path,"start":0,"end":1}]}}
  r = sess.post(f"{BASE}/render/start", json=payload)
  assert r.ok
  job_id = r.json()["job_id"]
  for _ in range(60):
    s = sess.get(f"{BASE}/render/status/{job_id}").json()
    if s["status"] == "done":
      return True
    if s["status"] == "failed":
      raise RuntimeError("render failed")
    time.sleep(2)
  raise RuntimeError("timeout")

if __name__ == "__main__":
  s = requests.Session()
  login(s)
  metrics(s)
  path = editor_upload(s)
  ok = render_job(s, path)
  print("SMOKE OK" if ok else "SMOKE FAIL")
