// Grab console element const consoleEl = document.getElementById("console"); // Utility: timestamp like [12:34:56] function stamp() { return new Date().toTimeString().slice(0, 8); } // Typewriter effect async function typeLine(line, delay = 20) { for (let i = 0; i < line.length; i++) { consoleEl.textContent += line[i]; await new Promise(r => setTimeout(r, delay)); } consoleEl.textContent += "\n"; } // Run intro boot log async function runLog() { const lines = [ `[${stamp()}] INCOMING HTTP REQUEST DETECTED ...`, `[${stamp()}] SERVICE WAKING UP ...`, `[${stamp()}] ALLOCATING RESOURCES ...`, `[${stamp()}] STARTING SYSTEM ...`, `[${stamp()}] ENVIRONMENT READY ...`, `[${stamp()}] INITIALIZING F.R.I.D.A.Y. PROTOCOL ...` ]; for (const l of lines) { await typeLine(l, 25); } const box = [ "┌──────────────────────────┐", "│ WELCOME TO F.R.I.D.A.Y │", "└──────────────────────────┘" ]; for (const b of box) { await typeLine(b, 2); } await typeLine(`[${stamp()}] SYSTEM ONLINE ✅`, 20); // After a short pause → go to dashboard setTimeout(() => window.location.href = "/home", 2500); } // Grid animation const gridEl = document.getElementById("grid"); for (let i = 0; i < 81; i++) { const d = document.createElement("div"); d.className = "cell"; gridEl.appendChild(d); } const cells = Array.from(gridEl.children); setInterval(() => { cells.forEach(c => c.classList.remove("active")); for (let i = 0; i < 5; i++) { cells[Math.floor(Math.random() * cells.length)].classList.add("active"); } }, 200); // Kick everything off runLog();
